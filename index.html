  public class DynamicObjectHub(GenericRepository genericRepository): Hub  {

    public async Task SendAsync(string clientMessage)  {

      var cancellationToken = CancellationToken.None;
      var dynamicObjectMessage = JsonConvert.DeserializeObject <DynamicObjectPropertyIds>
      (clientMessage);
      if (dynamicObjectMessage != null)  {

        var getDynamicObjectPropertiesTask = genericRepository.ExecuteFunctionAsync(
          StoredFunction.GetDynamicObjectProperties, dynamicObjectMessage,
          typeof (DynamicObjectPropertiesOut), cancellationToken);
        var dynamicObjectPropertiesLists = ((List<object>)(await getDynamicObjectPropertiesTask.ConfigureAwait(false) ?? new List<object>
          ())).OfType<DynamicObjectPropertiesOut>
        ();
        await this.Clients.Caller.SendAsync("ReceiveMessage", dynamicObjectPropertiesLists,
          cancellationToken).ConfigureAwait(false);
      }
    }
  }
